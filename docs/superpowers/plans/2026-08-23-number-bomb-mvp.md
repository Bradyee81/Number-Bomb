# Number Bomb MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Number Bomb MVP — a standalone Tauri desktop app with a two-window (operator control / chroma-key output) live number-guessing stage game, per the approved design.

**Architecture:** Vite + React 19 + TypeScript + Tailwind 4 frontend inside a Tauri 2 shell, two windows (`control`, `output`) split by window label, communicating over Tauri's event bus. Game round state is in-memory only (single-level undo via a snapshot). Settings (chroma color, explosion video path, voiceover volume) persist to a JSON file via `tauri-plugin-fs` — no database. Scaffold, digit-strip animation, chroma presets, and dual-window pattern are ported from the `Bradyee81/eventdraw` (Lucky Draw) reference app.

**Tech Stack:** Tauri 2, React 19, TypeScript, Tailwind CSS 4, Vite 7, Vitest (unit tests for game logic).

**Spec:** `docs/superpowers/specs/2026-08-23-number-bomb-design.md` (and `PRD-number-bomb.md` at repo root for full product rationale).

## Global Constraints

- Range is fixed at 01–99 for MVP — no configurable range, no configurable digit-strip count (PRD §2, §8).
- Digits are always displayed zero-padded to 2 characters (PRD §3).
- No out-of-range call validation, no repeat-call lockout list — the range logic self-enforces (PRD §2).
- Undo is single-level only — one snapshot, cleared once consumed (PRD §6, design §3).
- No player/roster tracking, no cross-round history, no scores (PRD §2, §8).
- Settings persist as a single JSON file via `tauri-plugin-fs` (app config dir) — no SQL/database dependency (design §2).
- Output window target resolution is 1920×1080 (PRD §9a; this differs from Lucky Draw's 1280×720 — do not copy that value).
- Explosion video reveal must never block the normal digit-reveal animation — video/red-flash is strictly additive (PRD §3, §5).
- Voiceover audio content (the actual `01.mp3`…`99.mp3` files) is explicitly out of scope for this plan — only the lookup-and-play plumbing is built (design §4, §7).
- `git push` is never part of an automated task step in this plan — commits are local only; pushing to `Bradyee81/Number-Bomb` happens separately, on explicit request.
- Tauri identifier: `com.brad.numberbomb`. Product name: `Number Bomb`.

- [ ] **Step 3: Write the implementation**

```ts
// src/lib/roundState.ts
export type RoundStatus = "active" | "exploded" | "ended";

export interface RoundState {
  bomb: number;
  low: number;
  high: number;
  status: RoundStatus;
}

export interface CallResult {
  state: RoundState;
  exploded: boolean;
}

export function randomBomb(low = 1, high = 99): number {
  return low + Math.floor(Math.random() * (high - low + 1));
}

export function startRound(bomb: number): RoundState {
  return { bomb, low: 1, high: 99, status: "active" };
}

export function endRound(state: RoundState): RoundState {
  return { ...state, status: "ended" };
}

export function callNumber(state: RoundState, n: number): CallResult {
  if (n === state.bomb) {
    return { state: { ...state, status: "exploded" }, exploded: true };
  }
  if (n < state.bomb) {
    return { state: { ...state, low: n }, exploded: false };
  }
  return { state: { ...state, high: n }, exploded: false };
}

export interface RoundReducerState {
  current: RoundState | null;
  undoSnapshot: RoundState | null;
}

export type RoundAction =
  | { type: "start"; bomb: number }
  | { type: "call"; n: number }
  | { type: "undo" }
  | { type: "end" };

export function roundReducer(state: RoundReducerState, action: RoundAction): RoundReducerState {
  switch (action.type) {
    case "start":
      return { current: startRound(action.bomb), undoSnapshot: null };
    case "call": {
      if (!state.current || state.current.status !== "active") return state;
      const { state: next } = callNumber(state.current, action.n);
      return { current: next, undoSnapshot: state.current };
    }
    case "undo":
      if (!state.undoSnapshot) return state;
      return { current: state.undoSnapshot, undoSnapshot: null };
    case "end":
      if (!state.current) return state;
      return { current: endRound(state.current), undoSnapshot: null };
    default:
      return state;
  }
}
```

```tsx
import { createContext, useCallback, useContext, useReducer, type ReactNode } from "react";
import {
  callNumber,
  randomBomb,
  roundReducer,
  startRound as computeStartRound,
  type RoundReducerState,
} from "../lib/roundState";
import { emitEvent } from "../lib/events";

const initialState: RoundReducerState = { current: null, undoSnapshot: null };

interface RoundContextValue {
  state: RoundReducerState;
  startRound: () => void;
  call: (n: number) => void;
  undo: () => void;
  endRound: () => void;
}

const RoundContext = createContext<RoundContextValue | null>(null);

export function RoundProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(roundReducer, initialState);

  const startRound = useCallback(() => {
    const bomb = randomBomb();
    const next = computeStartRound(bomb);
    dispatch({ type: "start", bomb });
    void emitEvent("bomb:round-start", { low: next.low, high: next.high });
  }, []);

  const call = useCallback(
    (n: number) => {
      if (!state.current || state.current.status !== "active") return;
      const { state: next, exploded } = callNumber(state.current, n);
      dispatch({ type: "call", n });
      void emitEvent("bomb:call", { n, low: next.low, high: next.high, exploded });
    },
    [state.current]
  );

  const undo = useCallback(() => {
    if (!state.undoSnapshot) return;
    const snapshot = state.undoSnapshot;
    dispatch({ type: "undo" });
    void emitEvent("bomb:undo", {
      low: snapshot.low,
      high: snapshot.high,
      exploded: snapshot.status === "exploded",
    });
  }, [state.undoSnapshot]);

  const endRound = useCallback(() => {
    if (!state.current) return;
    dispatch({ type: "end" });
    void emitEvent("bomb:round-end", {});
  }, [state.current]);

  return (
    <RoundContext.Provider value={{ state, startRound, call, undo, endRound }}>
      {children}
    </RoundContext.Provider>
  );
}

export function useRound(): RoundContextValue {
  const ctx = useContext(RoundContext);
  if (!ctx) throw new Error("useRound must be used within RoundProvider");
  return ctx;
}
```

```ts
import { useCallback, useRef, useState } from "react";
import type { ReelStatus } from "../components/SlotReel";
import { STAGGER_MS } from "../lib/reelTiming";

export type ReelKey = "lowTens" | "lowUnits" | "highTens" | "highUnits";
const REEL_KEYS: ReelKey[] = ["lowTens", "lowUnits", "highTens", "highUnits"];

function toTensUnits(n: number): [number, number] {
  const clamped = Math.max(0, Math.min(99, n));
  return [Math.floor(clamped / 10), clamped % 10];
}

export function useBoundReels() {
  const [statuses, setStatuses] = useState<Record<ReelKey, ReelStatus>>({
    lowTens: "locked",
    lowUnits: "locked",
    highTens: "locked",
    highUnits: "locked",
  });
  const [digits, setDigits] = useState<Record<ReelKey, number | null>>({
    lowTens: 0,
    lowUnits: 1,
    highTens: 9,
    highUnits: 9,
  });
  const timeoutsRef = useRef<Map<ReelKey, ReturnType<typeof setTimeout>>>(new Map());

  const clearPending = useCallback((key: ReelKey) => {
    const t = timeoutsRef.current.get(key);
    if (t) {
      clearTimeout(t);
      timeoutsRef.current.delete(key);
    }
  }, []);

  const animatePair = useCallback(
    (tensKey: ReelKey, unitsKey: ReelKey, value: number) => {
      const [tens, units] = toTensUnits(value);
      setDigits((prev) => ({ ...prev, [tensKey]: tens, [unitsKey]: units }));

      [tensKey, unitsKey].forEach((key, i) => {
        clearPending(key);
        const t = setTimeout(() => {
          timeoutsRef.current.delete(key);
          setStatuses((prev) => ({ ...prev, [key]: "landing" }));
        }, i * STAGGER_MS);
        timeoutsRef.current.set(key, t);
      });
    },
    [clearPending]
  );

  const setLow = useCallback((value: number) => animatePair("lowTens", "lowUnits", value), [animatePair]);
  const setHigh = useCallback((value: number) => animatePair("highTens", "highUnits", value), [animatePair]);

  const resetTo = useCallback(
    (low: number, high: number) => {
      animatePair("lowTens", "lowUnits", low);
      animatePair("highTens", "highUnits", high);
    },
    [animatePair]
  );

  const markLanded = useCallback((key: ReelKey) => {
    setStatuses((prev) => (prev[key] !== "landing" ? prev : { ...prev, [key]: "locked" }));
  }, []);

  return { statuses, digits, setLow, setHigh, resetTo, markLanded, keys: REEL_KEYS };
}
```

```tsx
import { useEffect, useRef, useState } from "react";

interface ExplosionOverlayProps {
  videoSrc: string;
  onDone: () => void;
}

export function ExplosionOverlay({ videoSrc, onDone }: ExplosionOverlayProps) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const [fallback, setFallback] = useState(false);

  useEffect(() => {
    const el = videoRef.current;
    if (!el) return;
    const handleEnded = () => onDone();
    const handleError = () => setFallback(true);
    el.addEventListener("ended", handleEnded);
    el.addEventListener("error", handleError);
    return () => {
      el.removeEventListener("ended", handleEnded);
      el.removeEventListener("error", handleError);
    };
  }, [onDone]);

  useEffect(() => {
    if (!fallback) return;
    const t = setTimeout(onDone, 1200);
    return () => clearTimeout(t);
  }, [fallback, onDone]);

  if (fallback) {
    return (
      <div className="pointer-events-none fixed inset-0 z-40 animate-[redFlash_0.25s_ease-in-out_4]" />
    );
  }

  return (
    // eslint-disable-next-line jsx-a11y/media-has-caption
    <video
      ref={videoRef}
```

```ts
import { useEffect, useState } from "react";
import { Window } from "@tauri-apps/api/window";
import { WebviewWindow } from "@tauri-apps/api/webviewWindow";

export interface PresentableWindow {
  isFullscreen(): Promise<boolean>;
  setFullscreen(fullscreen: boolean): Promise<void>;
  show(): Promise<void>;
  setFocus(): Promise<void>;
  listen(event: string, handler: () => void): Promise<() => void>;
}

async function getOrCreateOutputWindow(): Promise<PresentableWindow> {
  const existing = await Window.getByLabel("output");
  if (existing) return existing;
  const win = new WebviewWindow("output", {
    url: "index.html",
    title: "Number Bomb — Output",
    width: 1920,
    height: 1080,
  });
  await new Promise<void>((resolve, reject) => {
    win.once("tauri://created", () => resolve());
    win.once("tauri://error", (e) => reject(e));
  });
  return win;
}

export function usePresentToggle() {
  const [presenting, setPresenting] = useState(false);

  useEffect(() => {
    let unlisten: (() => void) | undefined;
    (async () => {
      const win = await Window.getByLabel("output");
      if (win) {
        unlisten = await win.listen("tauri://destroyed", () => setPresenting(false));
      }
    })();
    return () => unlisten?.();
  }, []);

  const toggle = async () => {
    const win = await getOrCreateOutputWindow();
    const isFull = await win.isFullscreen().catch(() => false);
    if (isFull) {
      await win.setFullscreen(false);
      setPresenting(false);
    } else {
      await win.show();
      await win.setFullscreen(true);
      await win.setFocus();
      setPresenting(true);
      await win.listen("tauri://destroyed", () => setPresenting(false));
    }
  };

  return { presenting, toggle };
}
```

```tsx
import { useState } from "react";
import { open } from "@tauri-apps/plugin-dialog";
import { copyFile, mkdir, BaseDirectory } from "@tauri-apps/plugin-fs";
import { CHROMA_PRESETS, type Settings } from "../../types";
import { useSettings } from "../SettingsContext";

export function SettingsScreen() {
  const { settings, updateSettings } = useSettings();
  const [busy, setBusy] = useState(false);

  if (!settings) return null;

  const set = (patch: Partial<Settings>) => {
    void updateSettings(patch);
  };

  const pickVideo = async () => {
    const selected = await open({
      multiple: false,
      filters: [{ name: "Video", extensions: ["mp4"] }],
    });
    if (!selected || Array.isArray(selected)) return;
    setBusy(true);
    try {
      await mkdir("media", { baseDir: BaseDirectory.AppData, recursive: true });
      const fileName = selected.split(/[\\/]/).pop() ?? "explosion.mp4";
      const destRelative = `media/${fileName}`;
      await copyFile(selected, destRelative, { toPathBaseDir: BaseDirectory.AppData });
      await updateSettings({ explosion_video_path: destRelative });
    } finally {
      setBusy(false);
    }
  };

  return (
    <div className="mx-auto max-w-2xl p-8">
      <h1 className="mb-1 text-xl font-bold">Settings</h1>

      <section className="mb-8">
        <label className="mb-2 block font-semibold">Chroma key background</label>
        <div className="grid grid-cols-4 gap-3">
          {CHROMA_PRESETS.map((preset) => (
            <button
              key={preset.hex}
              onClick={() => set({ chroma_color: preset.hex })}
              className={`flex flex-col items-center gap-1.5 rounded-md border p-2 ${
                settings.chroma_color === preset.hex
                  ? "border-[#F5C518] ring-1 ring-[#F5C518]"
                  : "border-white/10 hover:border-white/25"
              }`}
            >
              <div className="h-10 w-full rounded" style={{ backgroundColor: preset.hex }} />
              <span className="text-[11px] text-white/60">{preset.name}</span>
            </button>
          ))}
        </div>

        </p>
      </section>

      <section className="mb-8">

          {settings.explosion_video_path || "Using bundled default"}
```
